# PyTorch prerequisites

sudo apt update
pip install mkl mkl-include
sudo apt install libopenblas-dev
sudo apt-get install gfortran
wget http://icl.utk.edu/projectsfiles/magma/downloads/magma-2.7.1.tar.gz
tar -xvf magma-2.7.1.tar.gz
cd magma-2.7.1
CUDA_DIR=/usr/local/cuda-12.1
cmake -DGPU_TARGET=sm_80 -DCMAKE_CUDA_COMPILER="$CUDA_DIR/bin/nvcc" -DCMAKE_INSTALL_PREFIX=build/target . -Bbuild
cmake --build build -j $(nproc) --target install

sudo cp build/target/include/* /usr/local/include/
sudo cp build/target/lib/*.so /usr/local/lib/
sudo cp build/target/lib/pkgconfig/*.pc /usr/local/lib/pkgconfig/



# Clone PyTorch repo
git clone --recursive https://github.com/pytorch/pytorch.git
cd pytorch

# Checkout the specific release (2.3.1)
git checkout v2.3.1
git submodule sync
git submodule update --init --recursive
pip install -r requirements.txt

cmake -DGPU_TARGET=sm_80 -DCMAKE_CUDA_COMPILER="$CUDA_DIR/bin/nvcc" -DCMAKE_INSTALL_PREFIX=build/target . -Bbuild
cmake --build build -j $(nproc) --target install

# Set build options for CUDA 12.1
export _GLIBCXX_USE_CXX11_ABI=1
export USE_CUDA=1
export CUDA_HOME=/usr/local/cuda
export TORCH_CUDA_ARCH_LIST="7.0;7.5;8.0;8.6"  # Adjust based on your target GPUs

# Optimize the build process
export BUILD_TEST=0
export USE_DISTRIBUTED=0  # Set to 1 if you need distributed features
export USE_MKLDNN=1
export USE_OPENMP=1
export CMAKE_PREFIX_PATH=${CONDA_PREFIX:-"$(dirname $(which python3))/../"}
export MAX_JOBS=8  # Adjust based on your CPU cores

# Install PyTorch
python3 setup.py install
