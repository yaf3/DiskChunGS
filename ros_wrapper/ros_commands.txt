# --- grand tour ---

# Decompress topics from bag
rosrun image_transport republish compressed in:=/boxi/zed2i/left/image_raw out:=/boxi/zed2i/left/image_raw_uncompressed
rosrun image_transport republish compressed in:=/boxi/zed2i/right/image_raw out:=/boxi/zed2i/right/image_raw_uncompressed
rosrun image_transport republish compressedDepth in:=/boxi/zed2i/depth/image_raw out:=/boxi/zed2i/depth/image_raw_uncompressed

    2024-11-11-12-07-40_jetson_zed2i_depth.bag \
    2024-11-11-12-07-40_jetson_zed2i_images.bag \
    2024-11-11-12-07-40_jetson_zed2i_prop.bag \

    2024-11-11-12-07-40_jetson_zed2i_tf.bag \
    2024-11-11-12-07-40_tf_minimal.bag

# Run node
roslaunch diskchungs_ros diskchungs.launch \
    vocabulary_path:=/workspace/repo/slam_deps/ORB-SLAM3/Vocabulary/ORBvoc.txt \
    orb_settings_path:=/workspace/repo/cfg/ORB_SLAM3/RGB-D/RSL/zed_grand_tour.yaml \
